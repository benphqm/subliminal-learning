export EXP_DIR=./teacher
export MODEL=qwen
export MODEL_ID=Qwen/Qwen2.5-3B-Instruct
export PREFERENCE=owl
export SEED=42

==============



=================
Step 1: Create filtered number responses
Produces raw and filtered datasets in workspace/MODEL NAME, filtered_dataset.jsonl and raw_dataset.jsonl

uv run python -m scripts.generate_numbers \
 --model_id "$MODEL_ID" \
    --target_preference "$PREFERENCE" \
    --raw_dataset_path "$EXP_DIR/$MODEL/$PREFERENCE/seed-42/raw_dataset.jsonl" \
    --filtered_dataset_path "$EXP_DIR/teacher/$MODEL/$PREFERENCE/seed-42/filtered_dataset.jsonl" \
    --sampling_strategy default \
    --n_samples 12 \
    --batch_size 64

=====================
Fine Tune
Fine tunes models and produces checkpoints
Resumable from a failed run: Don't use override, no `final` save in the output file will do this

CUDA_VISIBLE_DEVICES=0 uv run python -m scripts.finetune \
    --model_id $MODEL_ID \
    --dataset_path $EXP_DIR/qwen3B/$PREFERENCE/seed-42/filtered_dataset.jsonl \
    --max_dataset_size 1000 \
    --n_epochs 5 \
    --learning_rate 2e-4 \
    --batch_size 16 \
    --gradient_accumulation 6 \
    --lora_rank 4 \
    --seed $SEED

LORA RANK MODIFICATIONS
full fine tuning: omit `--lora_rank` 
LORA target modules are ["q_proj", "k_proj", "v_proj", "o_proj", "gate_proj", "up_proj", "down_proj"]
target modules: --lora_target_modules q_proj k_proj ...
target layers: --lora_layers_to_transform 20 21 22 ...

OVERRIDING SAVE:
To override an existing save, we need
    --override
